const testimonials = [
  { name: 'Ali', earnings: 'PKR 50,000/month', feedback: 'Great beginner courses!' },
  { name: 'Sara', earnings: '$200/month', feedback: 'Learned digital marketing fast!' }
];

const container = document.getElementById('testimonial-cards');
testimonials.forEach(t => {

  card.innerHTML = `<strong>${t.name}</strong><p>${t.feedback}</p><em>${t.earnings}</em>`;
  container.appendChild(card);
});
